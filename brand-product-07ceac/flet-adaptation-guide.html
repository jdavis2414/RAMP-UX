import flet as ft

class ProductTheme:
    BG = "#ffffff"
    FG = "#111111"
    ACCENT = "#1677ff"
    SURFACE = "#f7f8fa"
    MUTED = "#6b7280"
    BORDER = "#d9dee7"
    RADIUS = 8

    @classmethod
    def apply(cls, page: ft.Page):
        page.bgcolor = cls.BG
        page.theme_mode = ft.ThemeMode.LIGHT
        
        # Load Inter dynamically from Google Fonts raw raw/static repository
        page.fonts = {
            "Inter": "https://raw.githubusercontent.com/google/fonts/main/ofl/inter/static/Inter-Regular.ttf",
            "Inter-Bold": "https://raw.githubusercontent.com/google/fonts/main/ofl/inter/static/Inter-Bold.ttf"
        }
        
        # Set primary color scheme defaults
        page.theme = ft.Theme(
            font_family="Inter",
            color_scheme=ft.ColorScheme(
                primary=cls.ACCENT,
                on_primary=cls.BG,
                surface=cls.SURFACE,
                on_surface=cls.FG,
                outline=cls.BORDER,
                background=cls.BG,
                on_background=cls.FG,
            )
        )
        page.update()
